#### 2.2.1 YieldMindCore.sol

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@openzeppelin/contracts-upgradeable/proxy/utils/UUPSUpgradeable.sol";
import "@openzeppelin/contracts-upgradeable/access/AccessControlUpgradeable.sol";
import "@openzeppelin/contracts-upgradeable/security/ReentrancyGuardUpgradeable.sol";
import "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import "@openzeppelin/contracts/token/ERC20/utils/SafeERC20.sol";

/**
 * @title YieldMindCore
 * @notice AI驱动的智能收益代理核心合约
 * @dev 使用 UUPS 代理模式实现可升级
 */
// 注：当前实现使用标准 OpenZeppelin (非升级模式)
// 以下为规划中的 UUPS 升级版本设计
contract YieldMindCore is
    UUPSUpgradeable,
    AccessControlUpgradeable,
    ReentrancyGuardUpgradeable 
{
    using SafeERC20 for IERC20;

    // ========== 角色定义 ==========
    bytes32 public constant STRATEGY_ROLE = keccak256("STRATEGY_ROLE");
    bytes32 public constant ORACLE_ROLE = keccak256("ORACLE_ROLE");
    bytes32 public constant GUARDIAN_ROLE = keccak256("GUARDIAN_ROLE");

    // ========== 状态变量 ==========
    
    /// @notice 支持的资产列表
    mapping(address => bool) public supportedAssets;
    
    /// @notice 用户存款: user => asset => amount
    mapping(address => mapping(address => uint256)) public userDeposits;
    
    /// @notice 用户策略分配: user => strategyId => asset => amount
    mapping(address => mapping(uint256 => mapping(address => uint256))) public userStrategyAllocations;
    
    /// @notice 策略TVL: strategyId => asset => amount
    mapping(uint256 => mapping(address => uint256)) public strategyTVL;
    
    /// @notice 用户累计收益
    mapping(address => mapping(address => uint256)) public userEarnings;
    
    /// @notice 累计协议收入
    mapping(address => uint256) public protocolRevenue;
    
    /// @notice 协议参数
    uint256 public protocolFeeRate; // 基点, 100 = 1%
    uint256 public dailyWithdrawalLimit;
    uint256 public maxStrategyAllocation; // 单策略最大分配比例 (基点)
    bool public emergencyPaused;
    
    /// @notice 全局计数器
    uint256 public totalStrategies;

    // ========== 事件 ==========
    
    event Deposited(address indexed user, address indexed asset, uint256 amount);
    event Withdrawn(address indexed user, address indexed asset, uint256 amount);
    event StrategyAllocated(address indexed user, uint256 indexed strategyId, address indexed asset, uint256 amount);
    event StrategyExecuted(uint256 indexed strategyId, address indexed asset, uint256 amount, uint256 profit);
    event EarningsClaimed(address indexed user, address indexed asset, uint256 amount);
    event EmergencyPaused(bool status);
    event RiskLimitBreached(uint256 indexed strategyId, string reason);

    // ========== 初始化 ==========
    
    /// @custom:oz-upgrades-unsafe-allow constructor
    constructor() {
        _disableInitializers();
    }

    function initialize(address admin, uint256 _feeRate) external initializer {
        __UUPSUpgradeable_init();
        __AccessControlUpgradeable_init();
        __ReentrancyGuardUpgradeable_init();
        
        _grantRole(DEFAULT_ADMIN_ROLE, admin);
        _grantRole(GUARDIAN_ROLE, admin);
        
        protocolFeeRate = _feeRate; // 默认 200 = 2%
        dailyWithdrawalLimit = 1000 * 1e18; // 默认日提取限额
        maxStrategyAllocation = 5000; // 单策略最大50%
    }

    // ========== 用户操作 ==========
    
    /**
     * @notice 存入资产
     * @param asset 资产地址
     * @param amount 存入数量
     */
    function deposit(address asset, uint256 amount) external nonReentrant {
        require(supportedAssets[asset], "Unsupported asset");
        require(amount > 0, "Amount must be > 0");
        require(!emergencyPaused, "Protocol paused");
        
        IERC20(asset).safeTransferFrom(msg.sender, address(this), amount);
        userDeposits[msg.sender][asset] += amount;
        
        emit Deposited(msg.sender, asset, amount);
    }

    /**
     * @notice 存入BNB并转换为WBNB
     */
    function depositBNB() external payable nonReentrant {
        require(!emergencyPaused, "Protocol paused");
        require(msg.value > 0, "Amount must be > 0");
        
        // 内部转换为WBNB
        address wbnb = 0xbb4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c;
        // ... WBNB deposit logic
        
        userDeposits[msg.sender][wbnb] += msg.value;
        emit Deposited(msg.sender, wbnb, msg.value);
    }

    /**
     * @notice 提取资产
     * @param asset 资产地址
     * @param amount 提取数量
     */
    function withdraw(address asset, uint256 amount) external nonReentrant {
        require(userDeposits[msg.sender][asset] >= amount, "Insufficient balance");
        
        // 检查日提取限额
        _checkWithdrawalLimit(msg.sender, amount);
        
        // 从策略中回收资金
        _unallocateFromStrategies(msg.sender, asset, amount);
        
        userDeposits[msg.sender][asset] -= amount;
        IERC20(asset).safeTransfer(msg.sender, amount);
        
        emit Withdrawn(msg.sender, asset, amount);
    }

    // ========== 策略操作 (仅 AI Oracle 可调用) ==========
    
    /**
     * @notice AI 执行策略分配
     * @dev 只有授权的 AI Oracle 可以调用
     * @param user 用户地址
     * @param strategyId 策略ID
     * @param asset 资产地址
     * @param amount 分配数量
     * @param signature AI 签名 (防篡改)
     */
    function executeStrategy(
        address user,
        uint256 strategyId,
        address asset,
        uint256 amount,
        bytes calldata signature
    ) external onlyRole(ORACLE_ROLE) nonReentrant {
        require(!emergencyPaused, "Protocol paused");
        require(strategyId < totalStrategies, "Invalid strategy");
        
        // 验证签名
        _verifyAISignature(user, strategyId, asset, amount, signature);
        
        // 验证分配比例不超过上限
        uint256 totalUserDeposits = userDeposits[user][asset];
        uint256 currentAllocation = userStrategyAllocations[user][strategyId][asset];
        require(
            (currentAllocation + amount) * 10000 / totalUserDeposits <= maxStrategyAllocation,
            "Exceeds max allocation"
        );
        
        // 执行策略
        userStrategyAllocations[user][strategyId][asset] += amount;
        strategyTVL[strategyId][asset] += amount;
        
        // 调用外部协议 (PancakeSwap/Venus等)
        _executeExternalStrategy(strategyId, asset, amount);
        
        emit StrategyAllocated(user, strategyId, asset, amount);
    }

    /**
     * @notice 记录策略收益
     * @param strategyId 策略ID
     * @param asset 资产地址
     * @param profit 收益金额
     */
    function recordProfit(
        uint256 strategyId,
        address asset,
        uint256 profit
    ) external onlyRole(STRATEGY_ROLE) {
        uint256 protocolFee = profit * protocolFeeRate / 10000;
        uint256 userProfit = profit - protocolFee;
        
        protocolRevenue[asset] += protocolFee;
        
        // 按比例分配给参与该策略的用户
        _distributeProfit(strategyId, asset, userProfit);
        
        emit StrategyExecuted(strategyId, asset, strategyTVL[strategyId][asset], profit);
    }

    /**
     * @notice 紧急暂停 (Guardian)
     */
    function emergencyPause(bool status) external onlyRole(GUARDIAN_ROLE) {
        emergencyPaused = status;
        emit EmergencyPaused(status);
    }

    // ========== 内部函数 ==========
    
    function _verifyAISignature(
        address user,
        uint256 strategyId,
        address asset,
        uint256 amount,
        bytes calldata signature
    ) internal view {
        bytes32 messageHash = keccak256(abi.encodePacked(
            user, strategyId, asset, amount, block.chainid, address(this)
        ));
        bytes32 ethSignedHash = messageHash.toEthSignedMessageHash();
        address signer = ethSignedHash.recover(signature);
        
        // 验证签名者是授权的 AI Oracle
        require(hasRole(ORACLE_ROLE, signer), "Invalid AI signature");
    }

    function _executeExternalStrategy(
        uint256 strategyId,
        address asset,
        uint256 amount
    ) internal {
        // 根据 strategyId 路由到不同的 DeFi 协议
        // 策略0: PancakeSwap V3 LP
        // 策略1: Venus 借贷
        // 策略2: Alpaca 杠杆挖矿
        // ...
    }

    function _unallocateFromStrategies(
        address user,
        address asset,
        uint256 amount
    ) internal {
        // 从各策略中按比例回收资金
    }

    function _distributeProfit(
        uint256 strategyId,
        address asset,
        uint256 profit
    ) internal {
        // 按用户在策略中的份额比例分配收益
    }

    function _checkWithdrawalLimit(address user, uint256 amount) internal view {
        // 检查日提取限额 (通过映射记录每日提取量)
    }

    function _authorizeUpgrade(address newImplementation) 
        internal 
        override 
        onlyRole(DEFAULT_ADMIN_ROLE) 
    {}
}
```

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@openzeppelin/contracts/access/AccessControl.sol";

/**
 * @title StrategyRegistry
 * @notice 策略注册表 - 管理所有可用策略
 */
contract StrategyRegistry is AccessControl {
    
    struct Strategy {
        uint256 id;
        string name;              // "PancakeSwap V3 BNB/USDT LP"
        string description;       // 策略描述
        address protocolAddress;  // 外部协议地址
        uint8 riskLevel;          // 1-5 风险等级
        uint256 minAPY;           // 最低APY (基点)
        uint256 maxAPY;           // 最高APY (基点)
        uint256 tvl;              // 总锁仓量
        bool isActive;            // 是否激活
        bool isDeprecated;        // 是否已弃用
        uint256 createdAt;
        uint256 lastUpdatedAt;
    }
    
    mapping(uint256 => Strategy) public strategies;
    uint256 public strategyCount;
    
    // 策略类型枚举
    enum StrategyType {
        LIQUIDITY_PROVISION,    // 流动性提供
        LENDING,                // 借贷
        LEVERAGED_FARMING,      // 杠杆挖矿
        YIELD_AGGREGATION,      // 收益聚合
        ARBITRAGE               // 套利
    }
    
    mapping(uint256 => StrategyType) public strategyTypes;
    
    event StrategyRegistered(uint256 indexed id, string name, uint8 riskLevel);
    event StrategyUpdated(uint256 indexed id, bool isActive);
    event StrategyDeprecated(uint256 indexed id);
    
    function registerStrategy(
        string calldata _name,
        string calldata _description,
        address _protocolAddress,
        StrategyType _type,
        uint8 _riskLevel,
        uint256 _minAPY,
        uint256 _maxAPY

            id: id,
            name: _name,
            description: _description,
            protocolAddress: _protocolAddress,
            riskLevel: _riskLevel,
            minAPY: _minAPY,
            maxAPY: _maxAPY,
            tvl: 0,
            isActive: true,
            isDeprecated: false,
            createdAt: block.timestamp,
            lastUpdatedAt: block.timestamp
        });
        
        strategyTypes[id] = _type;
        
        emit StrategyRegistered(id, _name, _riskLevel);
        return id;
    }
    
    function updateStrategyStatus(
        uint256 _id, 
        bool _isActive
    ) external onlyRole(DEFAULT_ADMIN_ROLE) {

        strategies[_id].lastUpdatedAt = block.timestamp;
        emit StrategyUpdated(_id, _isActive);
    }

        return strategies[_id];
    }

        uint256[] memory activeIds = new uint256[](strategyCount);
        uint256 count = 0;
        for (uint256 i = 0; i < strategyCount; i++) {
            if (strategies[i].isActive && !strategies[i].isDeprecated) {
                activeIds[count++] = i;
            }

        // 截取实际长度

        for (uint256 i = 0; i < count; i++) {
```

```solidity
            user: _user,
            strategyId: _strategyId,
            asset: _asset,
            amount: _amount,
            timestamp: block.timestamp,
            deadline: block.timestamp + executionWindow,
            signature: _signature,
            status: ExecutionStatus.Pending
        });
        
        emit AIExecutionSubmitted(requestId, _user, _strategyId);
        return requestId;
    }

     * @notice 执行待处理的AI决策
     */
    function executePending(bytes32 _requestId) external nonReentrant {
        AIExecution storage exec = executions[_requestId];
        
        require(exec.status == ExecutionStatus.Pending, "Not pending");
        require(block.timestamp <= exec.deadline, "Execution expired");
        require(
            block.timestamp >= lastExecutionTime[exec.user] + minExecutionInterval,
            "Too frequent"

        AIExecution storage exec = executions[_requestId];
        require(exec.status == ExecutionStatus.Pending, "Not pending");
        require(block.timestamp > exec.deadline, "Not expired");
        
        exec.status = ExecutionStatus.Expired;
        emit AIExecutionExpired(_requestId);

            exec.user, exec.strategyId, exec.asset, 
            exec.amount, exec.timestamp, block.chainid
        ));
        bytes32 ethSignedHash = messageHash.toEthSignedMessageHash();
        address signer = ethSignedHash.recover(exec.signature);
        require(authorizedAINodes[signer], "Invalid AI signature");
    }
```

```typescript
  async autoRebalance(userId: string): Promise<RebalanceResult> {

    const targetAllocation = await this.calculateOptimalAllocation(portfolio);
    
    const result = await this.agent.invoke({

请生成执行计划，包括：

  /**
   * 风险监控 (定时任务)
   */
  async monitorRisks(): Promise<RiskAlert[]> {
    const alerts: RiskAlert[] = [];
    
    // 1. 检查协议健康
    const protocolHealth = await this.checkProtocolHealth();
    if (protocolHealth.critical.length > 0) {
      alerts.push({
        level: "CRITICAL",
        type: "PROTOCOL_RISK",
        message: `检测到高风险协议: ${protocolHealth.critical.join(", ")}`,
        action: "IMMEDIATE_EXIT"
      });
    }

    // 2. 检查无常损失
    const ilStatus = await this.riskEngine.checkImpermanentLoss();
    for (const position of ilStatus.positions) {
      if (position.ilPercent > 5) {
        alerts.push({
          level: "HIGH",
          type: "IMPERMANENT_LOSS",
          message: `LP头寸 ${position.pair} 无常损失达到 ${position.ilPercent}%`,
          action: "CONSIDER_EXIT"
        });
      }
    }

    // 3. 检查价格异常
    const priceAlerts = await this.riskEngine.checkPriceAnomalies();
    alerts.push(...priceAlerts);

    // 4. 使用AI分析综合风险
    const aiAnalysis = await this.agent.invoke({
      input: `分析以下风险指标并提供应对建议：
${JSON.stringify(alerts, null, 2)}

    ]);

    return {

      gasPrice: gas.gasPrice,
      topPools,
      marketTrend,
      timestamp: Date.now()
    };
  }

  private async calculateOptimalAllocation(
```

```typescript
export interface StrategyMetrics {
  // 收益指标
  apy: number;              // 年化收益率
  apy7d: number;            // 7日年化
  apy30d: number;           // 30日年化
  
  // 风险指标
  volatility: number;       // 波动率
  sharpeRatio: number;      // 夏普比率
  sortinoRatio: number;     // 索提诺比率
  maxDrawdown: number;      // 最大回撤
  valueAtRisk: number;      // VaR (95%)
  
  // DeFi特定指标
  impermanentLoss: number;  // 无常损失
  tvl: number;              // 总锁仓量
  utilizationRate: number;  // 资金利用率
  
  // 综合评分
  safetyScore: number;      // 安全评分 (0-100)
  profitabilityScore: number; // 盈利评分 (0-100)
  overallScore: number;     // 综合评分 (0-100)
}

export class StrategyEvaluator {
  
  evaluate(pool: PoolData, historicalData: PriceData[]): StrategyMetrics {
    const returns = this.calculateReturns(historicalData);
    
    return {
      apy: this.calculateAPY(pool),
      apy7d: this.calculateAPY7d(pool),
      apy30d: this.calculateAPY30d(pool),
      
      volatility: this.calculateVolatility(returns),
      sharpeRatio: this.calculateSharpeRatio(returns),
      sortinoRatio: this.calculateSortinoRatio(returns),
      maxDrawdown: this.calculateMaxDrawdown(returns),
      valueAtRisk: this.calculateVaR(returns, 0.95),
      
      impermanentLoss: this.calculateIL(pool, historicalData),
      tvl: pool.tvl,
      utilizationRate: pool.utilizationRate,
      
      safetyScore: this.calculateSafetyScore(pool),
      profitabilityScore: this.calculateProfitabilityScore(pool),
      overallScore: 0 // 计算后填入
    };
  }

  private calculateSafetyScore(pool: PoolData): number {
    let score = 100;
    
    // TVL权重 (30%)
    if (pool.tvl < 1_000_000) score -= 30;
    else if (pool.tvl < 10_000_000) score -= 15;
    else if (pool.tvl < 50_000_000) score -= 5;
    
    // 审计状态 (25%)
    if (!pool.audited) score -= 25;
    
    // 运行时间 (15%)
    if (pool.ageDays < 30) score -= 15;
    else if (pool.ageDays < 90) score -= 10;
    else if (pool.ageDays < 180) score -= 5;
    
    // 治理去中心化 (15%)
    score -= (1 - pool.decentralizationScore) * 15;
    
    // 历史安全事件 (15%)
    score -= pool.securityIncidents * 5;
    
    return Math.max(0, Math.min(100, score));
  }

  private calculateSharpeRatio(returns: number[]): number {
    const avgReturn = returns.reduce((a, b) => a + b, 0) / returns.length;
    const riskFreeRate = 0.02 / 365; // 2% 年化 / 365天
    const volatility = this.calculateVolatility(returns);
    
    return (avgReturn - riskFreeRate) / volatility;
  }

  private calculateImpermanentLoss(
    pool: PoolData,
    priceChange: number
  ): number {
    // IL formula: 2 * sqrt(price_ratio) / (1 + price_ratio) - 1
    const ratio = 1 + priceChange;
    const il = 2 * Math.sqrt(ratio) / (1 + ratio) - 1;
    return Math.abs(il) * 100; // 返回百分比
  }
}
```

### 5.2 WebSocket 实时数据流 — 规划中 (Roadmap)

```typescript
// ============================================================
// 文件: src/api/websocket/handler.ts
// WebSocket 实时数据推送
// ============================================================

import { WebSocketServer, WebSocket } from "ws";
import { Redis } from "ioredis";

interface WSClient {
  ws: WebSocket;
  address: string;
  subscriptions: Set<string>;
}

export class YieldMindWebSocket {
  private wss: WebSocketServer;
  private clients: Map<string, WSClient> = new Map();
  private redis: Redis;

  constructor(port: number) {
    this.wss = new WebSocketServer({ port });
    this.redis = new Redis(process.env.REDIS_URL);
    this.setup();
  }

  private setup() {
    this.wss.on("connection", (ws, req) => {
      const clientId = this.generateId();
      const client: WSClient = {
        ws,
        address: "",
        subscriptions: new Set()
      };
      this.clients.set(clientId, client);

      ws.on("message", (data) => {
        const msg = JSON.parse(data.toString());
        this.handleMessage(clientId, msg);
      });

      ws.on("close", () => {
        this.clients.delete(clientId);
      });

      // 心跳
      const heartbeat = setInterval(() => {
        if (ws.readyState === WebSocket.OPEN) {
          ws.ping();
        } else {
          clearInterval(heartbeat);
        }
      }, 30000);
    });

    // Redis PubSub 订阅链上事件
    this.setupRedisSubscriptions();
  }

  private handleMessage(clientId: string, msg: any) {
    const client = this.clients.get(clientId);
    if (!client) return;

    switch (msg.type) {
      case "auth":
        client.address = msg.address;
        this.send(clientId, { type: "auth_success" });
        break;

      case "subscribe":
        client.subscriptions.add(msg.channel);
        this.send(clientId, { 
          type: "subscribed", 
          channel: msg.channel 
        });
        break;

      case "unsubscribe":
        client.subscriptions.delete(msg.channel);
        break;
    }
  }

  private setupRedisSubscriptions() {
    const sub = new Redis(process.env.REDIS_URL);

    // 订阅价格更新
    sub.subscribe("price:update");
    sub.subscribe("strategy:update");
    sub.subscribe("portfolio:update");
    sub.subscribe("alert:risk");
    sub.subscribe("transaction:confirmed");

    sub.on("message", (channel, message) => {
      const data = JSON.parse(message);
      
      // 推送给订阅了对应频道的用户
      this.clients.forEach((client, clientId) => {
        if (client.subscriptions.has(channel)) {
          // 如果是用户特定数据，只推送给该用户
          if (data.address && data.address !== client.address) return;
          this.send(clientId, { type: channel, data });
        }
      });
    });
  }

  private send(clientId: string, data: any) {
    const client = this.clients.get(clientId);
    if (client?.ws.readyState === WebSocket.OPEN) {
      client.ws.send(JSON.stringify(data));
    }
  }

  private generateId(): string {
    return Math.random().toString(36).substring(2, 15);
  }
}

/*
WebSocket 频道定义:

1. price:update - 代币价格实时更新
   数据: { token: string, price: number, change24h: number }

2. strategy:update - 策略APY变化
   数据: { strategyId: number, apy: number, tvl: number }

3. portfolio:update - 投资组合变化
   数据: { totalValue: number, dailyPnL: number, apy: number }

4. alert:risk - 风险警报
   数据: { level: string, type: string, message: string }

5. transaction:confirmed - 交易确认
   数据: { txHash: string, status: string, action: string }

6. ai:response - AI响应流
   数据: { sessionId: string, chunk: string, done: boolean }
*/
```
